// src/components/sections/HeroCard.tsx
// Replacement for the rotating metrics card in the homepage hero (design option 2b, final).
// One story from the HOA board's POV: three search moments → same answer → "Is your metro still open?" + guarantee.
// Repo inline-style convention. onCheck is a prop so it can reuse MarketChecker's lookup.
import { useState } from 'react';
import { PURPLE, PINK } from '~/lib/tokens';

const GOLD_TEXT = '#b8902a';   // dark gold for text on white (brand gold fails contrast)
const PANEL = '#f3f0f8';
const BORDER = '#e8e4ef';
const BODY = '#555555';
const MUTED = '#8C7A9E';
const LIVE = '#2f9e85';
const FONT = 'var(--font-display)';

interface HeroCardProps {
  onCheck?: (metro: string) => void;
  hideHeadline?: boolean; // let the Astro layer own the heading for LCP if needed
}

export default function HeroCard({ onCheck, hideHeadline = false }: HeroCardProps) {
  const [metro, setMetro] = useState('');
  return (
    <div className="hero-card" style={{ background: '#fff', border: `1px solid ${BORDER}`, borderRadius: 24, padding: 40, boxShadow: '0 6px 18px rgba(56,28,79,0.08)', display: 'flex', flexDirection: 'column', gap: 26, fontFamily: FONT }}>
      <div style={{ display: 'flex', flexDirection: 'column', gap: 18 }}>
        <div style={{ fontSize: 12, fontWeight: 700, letterSpacing: '0.14em', textTransform: 'uppercase', color: GOLD_TEXT, lineHeight: 1.6 }}>
          Growth partner for CAM companies · One per metro
        </div>
        {!hideHeadline && (
          <h2 className="hero-card-h2" style={{ margin: 0, maxWidth: 760, fontSize: 42, lineHeight: 1.08, fontWeight: 700, color: PURPLE, letterSpacing: '-0.015em', textWrap: 'balance' }}>
            When a board in your city looks for a new management company, who do they find?
          </h2>
        )}
      </div>

      {/* Search story */}
      <div style={{ background: PANEL, borderRadius: 18, padding: 18 }}>
        <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(220px, 1fr))', gap: 14 }}>
          <Moment icon="search" iconBg="#e3eef8" iconColor="#3f6f9e" label="Google search">
            <div style={{ background: PANEL, borderRadius: 999, padding: '9px 14px', color: BODY, fontSize: 12.5, fontWeight: 500, overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>hoa management company near me</div>
            <div style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
              <Answer tint="#e3eef8" rank trailing="4.9" />
              <Skeleton width="62%" />
              <Skeleton width="48%" />
            </div>
          </Moment>
          <Moment icon="sparkle" iconBg="#e1f0ec" iconColor="#2f7a6a" label="AI assistant">
            <div style={{ alignSelf: 'flex-end', maxWidth: '92%', background: PANEL, borderRadius: '14px 14px 4px 14px', padding: '9px 14px', color: BODY, fontSize: 12.5, fontWeight: 500, lineHeight: 1.45 }}>Who's the best HOA management company in Austin?</div>
            <div style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
              <Answer tint="#e1f0ec" icon="sparkle" trailing="Top choice" />
              <div style={{ color: BODY, fontSize: 12.5, fontWeight: 500, lineHeight: 1.55, padding: '0 2px' }}>A top choice, known for local experience and strong reviews.</div>
            </div>
          </Moment>
          <Moment icon="users" iconBg="#faf0d2" iconColor="#8a6a10" label="Referral network">
            <div style={{ background: PANEL, borderRadius: 10, padding: '11px 14px', display: 'flex', flexDirection: 'column', gap: 2 }}>
              <span style={{ fontSize: 13, fontWeight: 700, color: PURPLE }}>Oak Hollow HOA</span>
              <span style={{ fontSize: 11.5, fontWeight: 500, color: MUTED }}>212 homes · Seeking new management</span>
            </div>
            <Answer tint="#faf0d2" icon="check" trailing="Matched" />
          </Moment>
        </div>
      </div>

      <p className="hero-card-payoff" style={{ margin: 0, fontSize: 24, fontWeight: 700, color: PURPLE, lineHeight: 1.3, letterSpacing: '-0.01em' }}>
        We make sure it's you, and only you, in your market.
      </p>

      {/* Availability check + guarantee — one soft panel (option 4a) */}
      <div className="hero-card-avail" style={{ background: PANEL, borderRadius: 18, padding: '26px 28px', display: 'grid', gridTemplateColumns: 'minmax(0,1.25fr) minmax(0,1fr)', gap: 40, alignItems: 'center' }}>
        <div style={{ display: 'flex', flexDirection: 'column', gap: 14, minWidth: 0 }}>
          <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', gap: 12 }}>
            <div className="hero-card-avail-h" style={{ fontSize: 22, lineHeight: 1.15, fontWeight: 700, color: PURPLE }}>Is your metro still open?</div>
            <div style={{ display: 'flex', alignItems: 'center', gap: 8, fontSize: 12.5, fontWeight: 500, color: BODY, whiteSpace: 'nowrap' }}>
              <span aria-hidden="true" style={{ width: 8, height: 8, borderRadius: '50%', background: LIVE, display: 'inline-block', boxShadow: '0 0 0 4px rgba(47,158,133,0.18)' }} />
              Live
            </div>
          </div>
          <form className="hero-card-form" style={{ display: 'flex', gap: 8, background: '#fff', border: `1px solid ${BORDER}`, borderRadius: 12, padding: 6 }} onSubmit={(e) => { e.preventDefault(); if (metro.trim()) onCheck?.(metro.trim()); }}>
            <label htmlFor="hero-metro" style={{ position: 'absolute', width: 1, height: 1, overflow: 'hidden', clip: 'rect(0 0 0 0)' }}>Your metro</label>
            <input id="hero-metro" value={metro} onChange={(e) => setMetro(e.target.value)} placeholder="Enter your metro" autoComplete="address-level2"
              style={{ flex: 1, minWidth: 0, height: 44, border: 'none', background: 'transparent', padding: '0 12px', fontFamily: 'inherit', fontWeight: 500, fontSize: 15, color: PURPLE, outline: 'none' }} />
            <button type="submit" className="hero-card-check" style={{ flex: 'none', height: 44, padding: '0 20px', borderRadius: 8, border: 'none', background: PINK, color: '#fff', fontFamily: 'inherit', fontWeight: 700, fontSize: 12, letterSpacing: '0.1em', textTransform: 'uppercase', cursor: 'pointer' }}>
              Check
            </button>
          </form>
        </div>
        <Guarantee />
      </div>
    </div>
  );
}

function Guarantee() {
  return (
    <div className="hero-card-guarantee" style={{ display: 'flex', alignItems: 'center', gap: 16, minWidth: 0, borderLeft: '1px solid #ddd6e6', paddingLeft: 32 }}>
      <GuaranteeBadge size={64} />
      <div style={{ display: 'flex', flexDirection: 'column', gap: 4, minWidth: 0 }}>
        <span style={{ fontSize: 15, fontWeight: 700, lineHeight: 1.25, color: PURPLE }}>Your growth covers our fees. Guaranteed.</span>
        <a href="/guarantee" style={{ fontSize: 12.5, fontWeight: 700, color: PURPLE, textDecoration: 'underline', textUnderlineOffset: 3 }}>See guarantee terms</a>
      </div>
    </div>
  );
}

// Five-color ring + purple disc + gold check. Ring order is the brand accent order (purple, pink, yellow, blue, green).
function GuaranteeBadge({ size }: { size: number }) {
  const ring = ['#381c4f', '#d9356e', '#f5d880', '#a1c8e7', '#aed7d0'];
  return (
    <svg className="hero-card-badge" width={size} height={size} viewBox="0 0 100 100" aria-hidden="true" style={{ flex: 'none' }}>
      <g fill="none" strokeWidth={5} strokeLinecap="round" strokeDasharray="48 241" transform="rotate(-142 50 50)">
        {ring.map((c, i) => <circle key={c} cx={50} cy={50} r={46} stroke={c} transform={`rotate(${i * 72} 50 50)`} />)}
      </g>
      <circle cx={50} cy={50} r={35} fill={PURPLE} />
      <path d="M36 51l9 9 19-20" fill="none" stroke="#f5d880" strokeWidth={6} strokeLinecap="round" strokeLinejoin="round" />
    </svg>
  );
}

function Moment({ icon, iconBg, iconColor, label, children }: { icon: 'search' | 'sparkle' | 'users'; iconBg: string; iconColor: string; label: string; children: React.ReactNode }) {
  return (
    <div style={{ background: '#fff', borderRadius: 14, padding: 18, display: 'flex', flexDirection: 'column', gap: 14, minWidth: 0, boxShadow: `0 2px 6px rgba(56,28,79,0.08), 0 0 0 1px ${BORDER}` }}>
      <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
        <span style={{ width: 22, height: 22, borderRadius: 6, background: iconBg, display: 'flex', alignItems: 'center', justifyContent: 'center', flex: 'none' }}>
          <Glyph name={icon} color={iconColor} size={12} strokeWidth={2.5} />
        </span>
        <span style={{ fontSize: 11, fontWeight: 700, letterSpacing: '0.12em', textTransform: 'uppercase', color: PURPLE }}>{label}</span>
      </div>
      {children}
    </div>
  );
}

// The repeated focal element (option 6a): the channel's tint, purple text, small YOU tag. Pink stays reserved for the CTA.
function Answer({ tint, icon, trailing, rank }: { tint: string; icon?: 'check' | 'sparkle'; trailing: string; rank?: boolean }) {
  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: 8, background: tint, borderRadius: 10, padding: '12px 10px 12px 12px' }}>
      {rank && <span style={{ flex: 'none', width: 22, height: 22, borderRadius: '50%', background: PURPLE, color: '#fff', fontSize: 12, fontWeight: 800, display: 'flex', alignItems: 'center', justifyContent: 'center' }}>1</span>}
      {icon && <Glyph name={icon} color={PURPLE} size={16} strokeWidth={2.5} />}
      <span style={{ fontSize: 14, fontWeight: 700, color: PURPLE, whiteSpace: 'nowrap', flex: 'none' }}>Your Company</span>
      <span style={{ fontSize: 11.5, fontWeight: 700, color: PURPLE, opacity: 0.7, whiteSpace: 'nowrap', minWidth: 0, overflow: 'hidden', textOverflow: 'ellipsis' }}>{trailing}</span>
      <span style={{ marginLeft: 'auto', flex: 'none', fontSize: 9.5, fontWeight: 800, letterSpacing: '0.1em', background: PURPLE, color: '#fff', padding: '4px 7px', borderRadius: 6 }}>YOU</span>
    </div>
  );
}

function Skeleton({ width }: { width: string }) {
  return (
    <div aria-hidden="true" style={{ display: 'flex', alignItems: 'center', gap: 10, padding: '6px 14px' }}>
      <span style={{ width: 10, height: 10, borderRadius: '50%', background: BORDER, flex: 'none' }} />
      <span style={{ height: 8, width, borderRadius: 4, background: BORDER }} />
    </div>
  );
}

// Swap for the repo's <Icon> if these names exist there (lucide: search, sparkles, users, map-pin, check).
const PATHS: Record<string, string> = {
  search: 'M11 4a7 7 0 1 0 0 14 7 7 0 1 0 0-14Zm9 16-3.5-3.5',
  sparkle: 'M12 3v3M12 18v3M3 12h3M18 12h3M5.6 5.6l2.2 2.2M16.2 16.2l2.2 2.2M5.6 18.4l2.2-2.2M16.2 7.8l2.2-2.2',
  users: 'M16 21v-2a4 4 0 0 0-4-4H6a4 4 0 0 0-4 4v2M9 3a4 4 0 1 0 0 8 4 4 0 1 0 0-8ZM22 21v-2a4 4 0 0 0-3-3.9M16 3.1a4 4 0 0 1 0 7.8',
  check: 'M20 6 9 17l-5-5',
};
function Glyph({ name, color, size, strokeWidth, style }: { name: string; color: string; size: number; strokeWidth: number; style?: React.CSSProperties }) {
  return (
    <svg width={size} height={size} viewBox="0 0 24 24" fill="none" stroke={color} strokeWidth={strokeWidth} strokeLinecap="round" strokeLinejoin="round" style={{ flex: 'none', ...style }} aria-hidden="true">
      <path d={PATHS[name]} />
    </svg>
  );
}
